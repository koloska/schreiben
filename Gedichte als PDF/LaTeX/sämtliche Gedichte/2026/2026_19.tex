\begin{strophe}
\vers{Kleide dich an,}
\vers{oh Talisman.}
\vers{Eine Hose aus Glück}
\vers{und ein Hemd aus Zufriedenheit.}
\vers{Schütze dich vor den Strahlen,}
\vers{die dich in Traurigkeit baden.}
\vers{Eine Sonnenbrille, die Vernunft spendiert}
\vers{und Schuhe aus purer Freude.}
\vers{Kleide dich an,}
\vers{mein Talisman.}
\vers{Schütze dich vor den Strahlen,}
\vers{die dich Unzufrieden malen.}
\vers{Verwelke nicht,}
\vers{schütze dich vor dieser Hitze}
\vers{mit einer schmerzsicheren Mütze.}
\vers{Mein Talisman,}
\vers{bleibe mir bei,}
\vers{mein Herz kann nicht vollkommen sein,}
\vers{wenn du verbrennst, mein Vögelein.}
\end{strophe}
