\section{Gedichtsthema}
\stepcounter{ctrpoeminyear}
%\addcontentsline{toc}{section}{\thechapter.\arabic{ctrpoeminyear} \; Gedichtsthema}

\begin{strophe}
\vers{Nutzlos sitzt er in der Ecke}
\vers{und bewegt sich nicht vom Flecke,}
\vers{sucht in Gedanken auf und ab}
\vers{nach einem Thema, das ihm passt.}
\vers{Doch zu keinem Thema findet er Verse,}
\vers{was ihn nach einer Weile so sehr schmerzte,}
\vers{dass er begann davon zu schreiben}
\vers{wie nutzlos er die Zeit vertreibte.}
\vers{Er schrieb also Zeile für Zeile}
\vers{und es verblieb auf diese Weise}
\vers{ein Gedicht, dessen Thema es ist}
\vers{kein Thema zu finden für ein Gedicht}
\end{strophe}
