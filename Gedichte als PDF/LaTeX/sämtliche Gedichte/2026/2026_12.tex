\begin{strophe}
\vers{Jeden Tag gehe ich baden}
\vers{in einem See voll Dornen.}
\vers{Hänge am seidenen Faden}
\vers{und bin völlig verloren.}
\end{strophe}

\begin{strophe}
\vers{Mein Leben ist ein Paradies,}
\vers{in dem es von Tod nur so sprießt.}
\vers{Tod hier, Tod da,}
\vers{hier ein hops und tralala.}
\end{strophe}

\begin{strophe}
\vers{Nach den Bergen,}
\vers{zwischen den Sternen,}
\vers{lodert das Dunkel.}
\vers{Ach wie toll es dort funkelt.}
\end{strophe}

\begin{strophe}
\vers{Ich ziehe tausend Wägen}
\vers{voller Beton}
\vers{und kann kaum noch stehen,}
\vers{bin völlig verfrorn,}
\vers{doch trete ich weiter,}
\vers{Schritt für Schritt.}
\vers{Ich seh' schon das Licht,}
\vers{das auf mich wartet,}
\vers{am Ziel nur für mich.}
\end{strophe}

\begin{strophe}
\vers{Es ruft mich mit hundert Sirenen.}
\vers{Es schreit nach meiner Seele}
\vers{und weint leis' nach mir,}
\vers{ganz leis',}
\vers{so leis', dass jeder Atomare Zustandswechsel}
\vers{das Weinen übertönt.}
\end{strophe}

\begin{strophe}
\vers{Ich höre dich,}
\vers{ich höre dich rufen.}
\vers{Ich komme,}
\vers{ich will doch zu dir kommen.}
\end{strophe}

\begin{strophe}
\vers{Doch mich halten Stränge,}
\vers{Stränge aus tapferen Fäden,}
\vers{die nicht leicht reißen,}
\vers{sodass ich an Ort und Stelle kreise.}
\vers{Und kreise.}
\vers{Und kreise.}
\end{strophe}
