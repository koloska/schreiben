\section{Meine letzten Stunden auf Station}
\stepcounter{ctrpoeminyear}
%\addcontentsline{toc}{section}{\thechapter.\arabic{ctrpoeminyear} \;}

\begin{strophe}
\vers{Meine letzten Stunden auf Station.}
\vers{Hier ess' ich die letzte Ration,}
\vers{die mir Zuteil nach Willigkeit,}
\vers{dann ström' ich in Richtung Freiheit.}
\end{strophe}

\begin{strophe}
\vers{Ich gehe in den Lebenskampf,}
\vers{paar Nächte und paar Tage lang,}
\vers{bevor ich mich dem Todeskampf}
\vers{stelle, mit dem ich Jahre rang.}
\end{strophe}

\begin{strophe}
\vers{Bitte lasst mich friedlich gehen,}
\vers{haltet mich nicht auf, ob Regen,}
\vers{Wind oder Sonne die Atmosphäre prägen,}
\vers{denn ich möchte nicht mehr leben.}
\end{strophe}

\begin{strophe}
\vers{Meinen letzten Wunsch erfüll' ich mir}
\vers{indem ich dies Gedichtsband mir}
\vers{drucke für all jene,}
\vers{von denen ich bald egoistisch gehe.}
\end{strophe}

\begin{strophe}
\vers{Zuletzt will ich noch sagen,}
\vers{ich kann das Leben nicht ertragen}
\vers{-- obgleich die Welt 'ne Schöne ist --}
\vers{weil das Leben mich zutiefst auffrisst.}
\end{strophe}

\begin{strophe}
\vers{Drum statt das Leben mich fressen zu lassen,}
\vers{fresse ich das leben ums dabei zu belassen.}
\vers{Den Tod nehm' ich dann in Empfang,}
\vers{der mich seit jeher sucht und doch nie fand.}
\end{strophe}
