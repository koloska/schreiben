\section{Schmerzlich}
\stepcounter{ctrpoeminyear}
%\addcontentsline{toc}{section}{\thechapter.\arabic{ctrpoeminyear} \; Schmerzlich}

\begin{strophe}
\vers{Die Erde dreht sich rundherum}
\vers{und wieder ist ein Tag vergangen.}
\vers{Mal wieder hatt' ich nichts zu tun}
\vers{und versinke wieder in Gedanken.}
\end{strophe}

\begin{strophe}
\vers{Warum spielt mir dieses Leben}
\vers{immer diesen üblen Streich?}
\vers{Meine Muskeln fangen an zu beben}
\vers{und alle Gelenke werden weich.}
\end{strophe}

\begin{strophe}
\vers{All das wunderbar Schöne}
\vers{zieht an mir vorbei}
\vers{und in meiner Höhle}
\vers{ist kein Platz für zwei.}
\end{strophe}

\begin{strophe}
\vers{Ich bin allein}
\vers{in meinem Kopf}
\vers{und mach' mich klein,}
\vers{versink' im Loch.}
\end{strophe}

\begin{strophe}
\vers{Wo ist Sinn?}
\vers{Warum ich?}
\vers{Es ist schlimm.}
\vers{Bin lächerlich.}
\end{strophe}

\begin{strophe}
\vers{Immer}
\vers{merklich}
\vers{schlimmer.}
\vers{Schmerzlich.}
\end{strophe}
