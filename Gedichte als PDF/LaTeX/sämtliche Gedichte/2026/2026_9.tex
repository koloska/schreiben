\begin{strophe}
\vers{Sie sagte A und ich dann O.o}
\vers{Die Überraschung war so groß.}
\vers{Die Freude stieg zum Himmel auf}
\vers{und Kälte meinen Rücken auch.}
\vers{Ob es wohl wahr war, was sie sagte?}
\vers{Ein Tränchen aus dem Auge ragte.}
\vers{Ich schaute sie mit Skepsis an,}
\vers{doch Lächeln zog Gewissheit ran}
\vers{und so ward es mir klar wie nichts,}
\vers{sie brachte Wahrheit mir ans Licht.}
\end{strophe}

\begin{strophe}
\vers{Die Nachricht war wie ein Geschenk,}
\vers{denn wie ich weiter daran denk'}
\vers{so kommt mir der grobe Verdacht,}
\vers{dass sie die Lebzeit mir vervielfacht hat}
\vers{durch bloß die Worte, die sie sprach,}
\vers{als sie mir beichtete was naht.}
\vers{Grad eben wollte ich noch sterben,}
\vers{jetzt soll mein Leben länger werden}
\end{strophe}

\begin{strophe}
\vers{Ich wollte ihr die Freud' nicht nehmen}
\vers{und sagte ich würd' zu ihr stehen.}
\vers{Doch in mir flüstert mir mein Herz,}
\vers{dass all dies meinen Suizid}
\vers{nicht stoppt, sondern höchstens aufschiebt.}
\vers{Ich würd' so gern' ihr'm Plan treu bleiben,}
\vers{doch Todeswunsch konnt' mir nicht weichen.}
\vers{Wenn ich treu bliebe ihrem Schwur,}
\vers{so würde ich nicht Jahre nur --}
\vers{Nein, Jahrzehnte würd' ich weiterleben,}
\vers{bloß ihrer Überraschung wegen.}
\end{strophe}

\begin{strophe}
\vers{Ich versprach, dass ich's versuch',}
\vers{doch so fest hatte mich der Fluch,}
\vers{dass ich schon wusste als ich sprach,}
\vers{dass Lüge in den Worten lag,}
\vers{die ich ihr sanft um's Herze legte,}
\vers{dass meine Stimme ängstlich bebte.}
\vers{Ich ahnte nämlich, dass ich bald}
\vers{schon nicht mehr auf der Erde walt',}
\vers{sondern vom Himmel aus herab}
\vers{zu ihrer Überraschung klatsch',}
\vers{anstatt mit ihr auf Erden}
\vers{noch Jahre lang glücklich zu werden.}
\end{strophe}

\begin{strophe}
\vers{Verzeih mir, oh du gottgleich' Kind,}
\vers{du, die mir so viel Hoffnung singt,}
\vers{denn deinen Wunsch lass' ich in Scherben}
\vers{sobald ich's tue mit dem Sterben.}
\vers{Ich schulde dir zwar so viel Dank,}
\vers{doch leider bin ich so schwer krank,}
\vers{dass ich das Schicksal machen lass',}
\vers{was ich doch selber so sehr hass'.}
\vers{Ich möchte dich sehr unterstützen}
\vers{und könnte viele Jahre nützen,}
\vers{doch leider weiß ich selbst genau,}
\vers{dass ich mich nicht mehr lange trau'}
\vers{mir meinen Selbstmord aufzuschieben,}
\vers{obgleich wir uns wie Freunde lieben.}
\vers{Und so werde ich bald schon scheiden}
\vers{von dieser Welt für alle Zeiten}
\vers{bevor die Überraschung sprießt,}
\vers{die du mir liebevoll verrietst.}
\end{strophe}
