\section{Neue Ziele}

%\addcontentsline{toc}{section}{\thechapter.\arabic{ctrpoeminyear} \; Neue Ziele}

\begin{strophe}
\vers{Neue Ziele setz' ich mir}
\vers{damit ich nicht schon sterbe.}
\vers{Doch diese Ziele sind belanglos,}
\vers{drum fühl' ich mich erbärmlich.}
\vers{Welche Ziele kann ich haben}
\vers{damit ich Sinn in ihnen spüre?}
\vers{Täglich kämpf ich mit mir selbst,}
\vers{doch kann mich nicht beleben.}
\end{strophe}
