\begin{strophe}
\vers{Warum stirbst du wenn du weinst?}
\vers{Es ist doch Sommer -- Freudenzeit!}
\vers{Alle spielen lustig draußen,}
\vers{lassen es so richtig sausen.}
\vers{Warum sitzt du trüb daheim,}
\vers{lässt Dunkelheit in dir gedeih'n?}
\vers{Warum gibt es nichts weswegen}
\vers{du zu leben kannst erwegen}
\vers{allein des Leben Lebens wegen?}
\vers{So lässt du dich ständig gehen,}
\vers{tust in Richtung Tode streben.}
\vers{Höre auf dies' Netz zu weben,}
\vers{fange an selbst aufzustehen.}
\vers{Sonst sitzt du noch morgen da,}
\vers{leblos und dem Tode nah.}
\end{strophe}
