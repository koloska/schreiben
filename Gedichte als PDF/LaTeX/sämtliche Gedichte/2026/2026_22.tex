\begin{strophe}
\vers{Das geschriebene Wort,}
\vers{das gesprochene Lied.}
\vers{Ein verzweifelter Ort}
\vers{und doch hab ich's lieb.}
\end{strophe}

\begin{strophe}
\vers{Es trifft mitten ins Herz,}
\vers{endet mit einer Explosion,}
\vers{erfüllt dich mit Schmerz}
\vers{oder gleicht 'ner Vision.}
\end{strophe}

\begin{strophe}
\vers{Es ist meditativ,}
\vers{ruhig und still.}
\vers{In dich dringt es tief.}
\vers{Ein Kuss auf die Stirn.}
\end{strophe}

\begin{strophe}
\vers{Ein Kuss aus Liebe,}
\vers{Trauer, Gewalt,}
\vers{Glück oder Friede.}
\vers{So trägt es Gestalt.}
\end{strophe}

\begin{strophe}
\vers{Ein Gedicht ist mehr als seine Zeilen.}
\vers{Es umhüllt deine Seele und möchte verweilen.}
\vers{In einem Gedicht steckt mehr als Musik.}
\vers{Es ist wie ein Zauber, der dich zu sich rief.}
\end{strophe}
