\section{Dem Koch sein Tag}
\stepcounter{ctrpoeminyear}
%\addcontentsline{toc}{section}{\thechapter.\arabic{ctrpoeminyear} \; Dem Koch sein Tag}

\begin{strophe}
\vers{So viele Gäste sitzen um den Tisch.}
\vers{Essen, trinken, heute Fisch,}
\vers{morgen Fleisch, und gestern wars ne Suppe.}
\vers{Ich komm mir vor wie eine Puppe,}
\vers{die immerzu nur tanzt nach den Befehlen,}
\vers{die keinen Tag in meiner Küche fehlen.}
\end{strophe}

\begin{strophe}
\vers{Pausenzeiten gibt es keine,}
\vers{denn ständig tönt in diesem Heime}
\vers{der Wunsch nach einem guten Weine,}
\vers{der präsentiert sein muss bei Zeite.}
\vers{So laufe ich in weißem Kleide}
\vers{hin und her und blass wie Kreide.}
\end{strophe}

\begin{strophe}
\vers{Wann darf ich endlich schlafen gehen?}
\vers{Die Liebe ward mir nie geschehen.}
\vers{Ich lebe hier Tag ein Tag aus}
\vers{nur für den nächsten Schmaus,}
\vers{den all die Leute haben wollen}
\vers{um nicht den ganzen Tag zu schmollen.}
\end{strophe}
