\begin{strophe}
\vers{Wenn über deine Angelegenheiten}
\vers{immer andere entscheiden,}
\vers{wenn deinen ganzen Tagesablauf}
\vers{ständig jemand and'res aufbaut,}
\vers{wenn du um Freiheit zu ertragen}
\vers{musst jed's mal nach Erlaubnis fragen,}
\vers{dann lass dir klipp und klar mal sagen,}
\vers{dass dich Strukturen überragen.}
\end{strophe}

\begin{strophe}
\vers{Wenn du gestern, heute, morgen}
\vers{nicht mehr schaffst für dich zu sorgen,}
\vers{sind es genau diese Strukturen,}
\vers{die dich über Wasser zogen,}
\vers{als du beinahe grad ertrankst}
\vers{und tiefer richtung Abgrund sankst.}
\end{strophe}

\begin{strophe}
\vers{Doch halte nicht zu lang' das Händchen}
\vers{der Regeln und der and'ren Menschen,}
\vers{die rundum täglich dich versorgen}
\vers{und Ordnung in dei'm Leben formen.}
\vers{Denn verlierst du dich zu sehr darin,}
\vers{verlierst du deinen eig'nen Sinn}
\vers{nach selbstständiger Lebensweise}
\vers{und jeder Tag wird zu 'ner Reise}
\vers{durch Ängste über den Verlusst}
\vers{von Ordnung, die du jetzt gewusst.}
\end{strophe}

\begin{strophe}
\vers{Der Kampf in die Selbstständigkeit}
\vers{fällt sicherlich niemandem leicht}
\vers{Es ist ein Kampf gegen sich selbst.}
\vers{Die Frage ist ob du's aushälst.}
\vers{Ob du es schaffst die Willenskraft zu finden}
\vers{deiner Gemütlichkeit die Fesseln anzubinden,}
\vers{ob du es willst dich selbst zu retten}
\vers{indem du's schaffst deine Gedanken anzuketten}
\vers{die dich plagen, quälen und dir Schlechtes einreden}
\vers{und deinen Leib zu einer Puppe prägen,}
\vers{die kontrolliert wird von den Hexereien,}
\vers{die deine Gedanken ständig schreien.}
\end{strophe}

\begin{strophe}
\vers{Wenn du es schaffst mal loszulassen}
\vers{und die Strukturen dann verblassen}
\vers{und du die Freiheit kurz mal schnupperst}
\vers{und auf der Zunge die Selbstständigkeit recht knuspert,}
\vers{dann gehe Schtitt für Schritt so weiter}
\vers{und lasse zu dein' freudig' Eifer}
\vers{nach einem Leben, das du selbst}
\vers{auf eig'nen Beinen dir erstellst.}
\end{strophe}

\begin{strophe}
\vers{Denn nur dann bist du selbst Struktur,}
\vers{wenn du Struktur fallen lässt}
\vers{und deinen eig'nen Weg betritst.}
\end{strophe}
