\begin{strophe}
\vers{Hey, das hier ist kein Abschiedsbrief,}
\vers{es ist ein Gruß, den ich dir schrieb,}
\vers{um mittels meiner letzten Zeilen}
\vers{ein letztes Tschüss mit dir zu teilen.}
\end{strophe}

\begin{strophe}
\vers{Ich danke dir für all die Zeit,}
\vers{die wir hatten vereint zu zweit.}
\vers{Gemeinsam haben wir geerntet}
\vers{so ach viele schöne Momente.}
\end{strophe}

\begin{strophe}
\vers{Ich möchte es dir gern verraten:}
\vers{Es freut mich dich gekannt zu haben,}
\vers{denn oft hab' ich durch dich erkannt}
\vers{wie schön ich doch das Leben fand.}
\end{strophe}

\begin{strophe}
\vers{Mit dir an meiner Seite}
\vers{durchlief ich meines Lebens Reise.}
\vers{Und dir wünsche ich weiterhin}
\vers{ein langes Leben voller Sinn!}
\end{strophe}

\begin{strophe}
\vers{Ich, leider, möchte nicht mehr leben,}
\vers{drum muss ich von dir Abschied nehmen.}
\vers{Bemitleide mich bitte nicht,}
\vers{denn ich entscheid' mich absichtlich.}
\end{strophe}

\begin{strophe}
\vers{Vor vielen Jahren schon}
\vers{wollte ich sterben ohne Ton,}
\vers{doch damals war ich zu K.O.}
\vers{um durchzuführen meinen Tod.}
\vers{Seit diesem Tag versprach ich mir}
\vers{dass ich es doch nochmal probier}
\vers{zu leben für das kleine Glück}
\vers{statt mich zu töten -- wie verrückt!}
\vers{Das klappte viele Jahre lang}
\vers{bis vor paar Wochen, als ich sank}
\vers{in ein so tiefes dunkles Loch}
\vers{in dem es stark nach Selbstmord roch.}
\vers{Dieser Gestank sucht mich jetzt heim}
\vers{und fordert mein Versprechen ein.}
\end{strophe}

\begin{strophe}
\vers{Ich bin recht froh über die Jahre,}
\vers{die ich mir gesammelt habe.}
\vers{Doch immer war mir eines klar,}
\vers{und zwar dass mir ein Ende naht,}
\vers{bei dem ich selbst der Täter bin}
\vers{und mich doch auch als Opfer find'.}
\vers{Das Datum war mir nie bekannt,}
\vers{doch scheinbar ist es jetzt so weit.}
\end{strophe}

\begin{strophe}
\vers{Drum lange Rede kurzer Sinn,}
\vers{ich muss mich jetzt verabschieden.}
\vers{Auf eig'nen Wunsch gehe ich fort,}
\vers{drum trauert nicht um meinen Mord.}
\end{strophe}

\begin{strophe}
\vers{Auf lange Zeit, auf nimmer wieder sehen.}
\vers{Ich weiß dass ich dich sehr verletze}
\vers{mit meinem Selbstmord,}
\vers{aber ich kann nicht anders,}
\vers{also tschüss,}
\vers{ich hab dich lieb! <3}
\end{strophe}
