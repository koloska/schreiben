\section{Kleiner Mann}
\stepcounter{ctrpoeminyear}

\begin{strophe}
\vers{Ich bin nur ein kleiner Mann,}
\vers{der nichts groß verändern kann.}
\vers{Was ich schreib' wird untergehn,}
\vers{Niemals wirds die Masse sehn.}
\vers{Warum schreib' ich also dann?}
\vers{Weil ich klein verändern kann?}
\vers{Meine Liebsten, meine Nächsten.}
\vers{Nehme Einfluss auf der'n Leben.}
\end{strophe}

\begin{strophe}
\vers{Leider schreibe ich nur Sachen,}
\vers{die niemandem Freude machen.}
\vers{Sollt' ich also weiter schreiben?}
\vers{Würd' nur Leid weiter verbreiten.}
\vers{Klingt nach 'ner schlechten Idee,}
\vers{doch leider hilft es mir auch sehr,}
\vers{am Leben weiter festzuhalten.}
\vers{Drum, sollt' ich, um auf Erd zu bleiben,}
\vers{weiter meine Zeilen schreiben,}
\vers{weiterhin Gedichte zeichnen?}
\end{strophe}

\begin{strophe}
\vers{Kaum wer will mich sterben seh'n,}
\vers{doch leider will ich von hier geh'n.}
\vers{Ständig leben tut mir Leid,}
\vers{doch auch zu sterben wäre weit}
\vers{entfernt von dem, was viele}
\vers{wünschten, ich machte mir zum Ziele.}
\end{strophe}

\begin{strophe}
\vers{Also schreib' ich hin und wieder,}
\vers{all diese Geschichten nieder.}
\vers{Doch warte ich auf einen Tag,}
\vers{an dem ich nicht mehr leben mag,}
\vers{Und mir entgültig einen Strich}
\vers{ziehe duch die Lebenspflicht.}
\end{strophe}
