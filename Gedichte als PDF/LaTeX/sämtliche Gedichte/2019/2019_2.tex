\section{Wir Soldaten}
\stepcounter{ctrpoeminyear}
%\addcontentsline{toc}{section}{\thechapter.\arabic{ctrpoeminyear} \; Wir Soldaten}

\begin{strophe}
\vers{Wir sind im Krieg,}
\vers{nur für den Sieg.}
\vers{Wir nehmen jeden Schuss,}
\vers{doch kriegen keinen Kuss.}
\vers{Wir rauchen Zigaretten,}
\vers{und schlafen ohne Betten.}
\vers{Wir fahren Panzer,}
\vers{für uns'ren Kanzler.}
\vers{Wir sind die Killer,}
\vers{nach uns wird alles stiller.}
\vers{Wir sind Soldaten,}
\vers{wir werden hier begraben.}
\end{strophe}
