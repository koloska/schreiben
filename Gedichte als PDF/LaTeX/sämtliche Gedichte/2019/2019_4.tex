\section{Der Heckenschlüpfer}
\stepcounter{ctrpoeminyear}
%\addcontentsline{toc}{section}{\thechapter.\arabic{ctrpoeminyear} \; Der Heckenschlüpfer}

\begin{strophe}
\vers{Er schlüpfte da,}
\vers{er schlüpfte dort,}
\vers{so blieb er nie an einem Ort.}
\vers{Es war ihm klar!}
\vers{Er wollte weg,}
\vers{wollte nicht bleiben,}
\vers{keine Zeit verstreichen!}
\vers{Ab in die nächste Heck'!}
\vers{Morgen schon wieder wo anders}
\vers{als er gestern war.}
\vers{Er mag es, klar.}
\vers{So oft verwandert.}
\vers{Fort ist er gern,}
\vers{schaut alles an,}
\vers{ganz in dem Bann,}
\vers{bleibt er mal stehn.}
\vers{Doch ist er ganz schnell wieder fort}
\vers{und schlüpft herum,}
\vers{ganz leis', ganz stumm,}
\vers{so bleibt er nie an einem Ort.}
\end{strophe}
