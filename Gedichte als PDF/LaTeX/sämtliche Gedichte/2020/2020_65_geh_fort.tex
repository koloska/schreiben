\subsection*{Geh}

\begin{strophe}
\vers{Bevor du gehst}
\vers{sag mir ob du verstehst,}
\vers{was ich dir sagen will?}
\vers{Dir ganz allein und still.}
\end{strophe}

\begin{strophe}
\vers{Bevor du gehst}
\vers{Lass mich wissen ob du verstehst,}
\vers{wieso ich geh?}
\vers{Es tut ja so weh.}
\end{strophe}

\subsection*{Fort}

\begin{strophe}
\vers{Wo bist du gewesen}
\vers{als ich begann zu verwesen?}
\vers{Weißt du wo ich nun bin?}
\vers{Ich bin jetzt tot.}
\end{strophe}
