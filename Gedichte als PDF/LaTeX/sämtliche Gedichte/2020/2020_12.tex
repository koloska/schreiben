%\addcontentsline{toc}{section}{\thechapter.\arabic{ctrpoeminyear}\; Albtraum}

\begin{strophe}
\vers{Wenn um Mitternacht der Schlaf beginnt,}
\vers{ganz tiefe träumt das kleine Kind,}
\vers{wenn die Eltern sich im Bett bewegen,}
\vers{Wind und Donner an der Scheibe beben,}
\vers{wenn draußen alles Dunkel singt,}
\vers{sich das Lied des Albtraums um ihn ringt,}
\vers{das Kindchen zappelt, hin und her,}
\vers{dann hat das Kindchen es sehr schwer.}
\end{strophe}
