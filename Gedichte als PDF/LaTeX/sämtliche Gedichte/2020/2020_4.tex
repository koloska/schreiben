\section{Blauer Fleck}
\stepcounter{ctrpoeminyear}
%\addcontentsline{toc}{section}{\thechapter.\arabic{ctrpoeminyear} \; Blauer Fleck}

\begin{strophe}
\vers{Blau ist der Fleck}
\vers{und will nicht mehr weg.}
\vers{So sehr ich ihn bitte,}
\vers{er bleibt in der Mitte.}
\vers{So sehr ich auch belle,}
\vers{er bleibt an der Stelle.}
\vers{Blau ist der Fleck.}
\vers{Ich wünscht ihn mir weg.}
\end{strophe}
