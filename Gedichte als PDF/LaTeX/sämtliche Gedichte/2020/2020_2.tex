\section{Das Loch}
\stepcounter{ctrpoeminyear}
%\addcontentsline{toc}{section}{\thechapter.\arabic{ctrpoeminyear} \; Das Loch}

\begin{strophe}
\vers{Frühmorgens auf dem Felde,}
\vers{wie die Weite sich fernte,}
\vers{lief ein kleines Mädchen fein}
\vers{am Wegesrand bis in den Wald hinein.}
\end{strophe}

\begin{strophe}
\vers{Dort wartete schon der Nebel,}
\vers{der sie sogleich verschluckte}
\vers{und das unschuldige Mädel}
\vers{in ein tiefes Loch reinspuckte.}
\end{strophe}

\begin{strophe}
\vers{Bevor sie es merkte,}
\vers{schwand ihre Stärke.}
\vers{Bevor sie verstand,}
\vers{verschwand ihr Verstand.}
\end{strophe}
