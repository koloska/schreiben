\section{Der Welt ein Ende}
\stepcounter{ctrpoeminyear}
%\addcontentsline{toc}{section}{\thechapter.\arabic{ctrpoeminyear} \; Der Welt ein Ende}

\begin{strophe}
\vers{Grausig, sausig, richtig toll!}
\vers{Schlaflos alles wider quoll.}
\vers{Ohne Worte der Worte Welt,}
\vers{der Welt ein Ende wurde bestellt.}
\vers{Der Engel, der singet er habe genug,}
\vers{verbietet bellend den wideren Spuk.}
\vers{Doch es nimmt kein Ende! Alles ist fort.}
\vers{Möcht er noch singen, so singt er den Tod.}
\end{strophe}
