\begin{strophe}
\vers{Wie ich aus dem Fenster schaue,}
\vers{und die Kinder erblicke,}
\vers{so wird mir warm ums Herz.}
\vers{Doch wenn ich daran denke,}
\vers{wenn ich träume,}
\vers{albträume}
\vers{über die Zukunft}
\vers{der Kinder.}
\vers{Dann komme ich um.}
\vers{Mein Herz erlischt.}
\end{strophe}
