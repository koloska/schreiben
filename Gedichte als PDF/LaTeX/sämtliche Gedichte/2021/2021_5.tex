\begin{strophe}
\vers{Liebste Menschen die ich liebe, erkennt meinen Schmerz.}
\vers{Verzeiht meine Absicht, es brennt mir das Herz.}
\vers{Es brennt wie ein Zündstab, es brennt lichterloh.}
\vers{Es wird bald erlischen, vergebt mein Fauxpas.}
\end{strophe}

\begin{strophe}
\vers{Schon bald bin ich fort, ich ertrag's hier nicht mehr.}
\vers{Ich würd gern noch bleiben, ich wünscht es mir sehr.}
\vers{Doch den Wunsch zu stillen bringt Leid meinem Leib.}
\vers{Drum geh ich allein, bis ich allein nicht mehr bleib.}
\end{strophe}

\begin{strophe}
\vers{Ich bitte euch Freunde, bloß weint nicht nach mir!}
\vers{Seid glücklich auf Erden und ich glücklich hier.}
\end{strophe}

\vers{Nun überlass ich mein Schicksal der Schicksal Hand}
\vers{Und hoffe den Tod zu empfangen bald.}
\end{strophe}
