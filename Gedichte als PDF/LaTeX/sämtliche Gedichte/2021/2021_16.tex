\section{In Karo}
\stepcounter{ctrpoeminyear}
%\addcontentsline{toc}{section}{\thechapter.\arabic{ctrpoeminyear}\; In Karo}

\begin{strophe}
\vers{Das karierte Muster auf ihrer Brust.}
\vers{Als ich es sah}
\vers{kam ich um,}
\vers{denn es ertönten sogleich Sirenen}
\vers{von der Straße nebenan.}
\vers{Sie wusste sie stirbt auf der Stelle.}
\vers{Wie sie weiterlief}
\vers{erklärte sich mir nie.}
\vers{Sie lächelte dem Tod entgegen,}
\vers{der sie beschoss.}
\vers{Aus zwanzig Kanonen flogen Patronen und Hülsen.}
\vers{Durchlöchert fiel sie auf den kalten Asphalt,}
\vers{der sofort zu brennen begann.}
\vers{Alle Passanten verflossen.}
\vers{Mit offenen Mündern streiften sie weiter.}
\vers{Das tote Mädchen schwebte den Himmel auf.}
\vers{Ihr Blut hüllte die Welt}
\vers{für einen Moment -}
\vers{Alles zersprang, alles warf sich, alles explodierte.}
\vers{Die Menschen starben.}
\vers{Die Menschen gewannen ihre erste Schlacht.}
\end{strophe}
