\section{Ich wünschte ich wäre tot}
\stepcounter{ctrpoeminyear}
%\addcontentsline{toc}{section}{\thechapter.\arabic{ctrpoeminyear}\; Ich wünschte ich wäre tot}

\begin{strophe}
\vers{Ich wünschte ich wäre tot}
\vers{dann könnte ich schlafen}
\vers{in aller ruhe und mich erholen}
\vers{von täglichem tode}
\vers{und weinen alleine}
\vers{ungestört sein}
\vers{beim malen}
\vers{von tausend regalen}
\vers{die tragen tausend beschwerden}
\vers{was wird nur aus mir werden}
\vers{ich frage mich wieder}
\vers{wieso ich nicht springe}
\vers{denn darin liegt friede.}
\end{strophe}
