\section{Was ich mache ist nicht dichten}
\stepcounter{ctrpoeminyear}
%\addcontentsline{toc}{section}{\thechapter.\arabic{ctrpoeminyear}\; Was ich mache ist nicht dichten}

\begin{strophe}
\vers{Was ich mache ist nicht dichten.}
\vers{Ich erfinde nur Geschichten,}
\vers{und schreibe sie dann auf,}
\vers{in Versformat, worauf}
\vers{ich mich dann umbringen will,}
\vers{weil das ist hier doch kein Spiel!}
\end{strophe}
