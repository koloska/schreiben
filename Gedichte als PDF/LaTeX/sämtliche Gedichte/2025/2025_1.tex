\stepcounter{ctrpoeminyear}

\begin{strophe}
\vers{Wann habe ich verlernt zu schreiben?}
\vers{Würd gern ein paar blanke Zeilen}
\vers{schmücken mit all jenen Worten}
\vers{die mir damals immer kamen}
\end{strophe}

\begin{strophe}
\vers{als ich Gefühlen einen Ausdruck gab,}
\vers{wie Emotionen sich zu deutschen Worten}
\vers{sich wie von selbst manifestierten.}
\end{strophe}

\begin{strophe}
\vers{Warum fiel mir das mal so einfach}
\vers{als ich noch jung war}
\vers{konnte ich so viel mehr als jetzt noch.}
\end{strophe}

\begin{strophe}
\vers{Vielleicht soll ich einfach merken,}
\vers{dass meine besten Tage rum sind,}
\vers{dass mein Hirn schlechter wird}
\twolinevers{und ich kein junger Gott mehr bin sondern nur noch}{erwachsen.}
\end{strophe}
