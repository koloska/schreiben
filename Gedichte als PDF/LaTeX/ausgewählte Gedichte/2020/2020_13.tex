%\addcontentsline{toc}{section}{\thechapter.\arabic{ctrpoeminyear}\; Der Bücher Leben}

\begin{strophe}
\vers{Voll Worte stehn sie im Regal.}
\vers{Zig Tausende. Allein die Zahl!}
\vers{Ob sie wohl wissen was sie sagen?,}
\vers{dort inmitten ihrer Paragraphen?,}
\vers{die immerzu mein Herze trafen,}
\vers{alsbald ich froh in Händen hatte}
\vers{ihre Blätter, werfend tausend Schatten!,}
\vers{ihre Seiten, vollkommen, wie sie waren.}
\end{strophe}

\begin{strophe}
\vers{Wer schrieb sie?, dieser Texte Weichen,}
\vers{wie wunderschön sie waren..., ohnegleichen!}
\vers{Wer war es, der gottgleich erschuf?,}
\vers{das hierige, das dort'ge Buch?}
\end{strophe}

\begin{strophe}
\vers{Jedes summte schöne Lieder,}
\vers{gab hübsche Melodien wieder.}
\vers{So blieben sie an ihrem Platze,}
\vers{tanzend, bis sie jemand fasste.}
\end{strophe}
