\begin{strophe}
\vers{Ich kann heut nicht dichten,}
\vers{es geht heute nicht.}
\vers{Was ich versuche,}
\vers{es klappt einfach nicht.}
\vers{Der Wunsch allein,}
\vers{er zählt nun mal nicht.}
\vers{Wenn sich die Reime nicht bilden,}
\vers{dann quälen sie mich.}\end{strophe}
