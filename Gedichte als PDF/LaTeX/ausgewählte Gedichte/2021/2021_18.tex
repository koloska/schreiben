\section{Die gute Nutte}
\stepcounter{ctrpoeminyear}
%\addcontentsline{toc}{section}{\thechapter.\arabic{ctrpoeminyear}\; Die gute Nutte}

\begin{strophe}
\vers{Und sie fragt mich, diese Nutte,}
\vers{ob ich etwas brauche, wenn es mir nicht gut geht.}
\vers{Und dann bitte ich, noch mehr Schläge, diese Nutte.}
\vers{Was für ein gutes Kerlchen du doch bist, lobt sie,}
\vers{schlägt mich, was für ein guter Sohn du bist!}
\vers{Zwickt mich!}
\vers{Die gute Nutte!}
\end{strophe}
