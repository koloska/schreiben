\section{Wann hast du zuletzt}
\stepcounter{ctrpoeminyear}
%\addcontentsline{toc}{section}{\thechapter.\arabic{ctrpoeminyear}\; Wann hast du zuletzt}

\begin{strophe}
\vers{Wann hast du zuletzt}
\vers{Ein wenig Glück empfunden?}
\vers{Schaue auf deine Wunden}
\vers{Und sei nicht verletzt.}
\vers{Auf so eine einfache Frage}
\vers{Antwortest du nein?}
\vers{Grabe in dich hinein,}
\vers{Und jetzt noch mal sage}
\vers{Mir die Wahrheit.}
\vers{An welchem Tage}
\vers{War gut deine Lage?}
\vers{Schaffe mir Klarheit,}
\vers{Ob gestern, letzte Woche}
\vers{Oder schon Jahre nicht mehr?}
\vers{Es interessiert mich nicht sehr,}
\vers{Doch grabe im Loche}
\vers{Deiner Erinnerungen.}
\vers{Sage mir, Herzchen,}
\vers{Ich fühl doch dein Schmerzchen.}
\vers{Doch hast du es überwunden}
\vers{Dann erzähle mir bitte}
\vers{Ob du jemals in einem Moment}
\vers{Mit Glücklichkeit warst beschenkt?}
\vers{Oder weshalb sind die Schnitte?}
\end{strophe}
