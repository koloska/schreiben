\begin{strophe}
\vers{Zu zweit nebeneinander zu zweit,}
\vers{zufällig, doch auch nicht weit,}
\vers{nur ein bisschen Tat war dabei,}
\vers{sonst zufällig, drum kein Geschrei.}
\end{strophe}

\begin{strophe}
\vers{Der weite Weg vom stummen Mund weg}
\vers{gelingt und schon wird geredet,}
\vers{doch nur ganz kurz, ein hin und her,}
\vers{kein Thema, das die Welt bewegt.}
\end{strophe}

\begin{strophe}
\vers{Doch ist nicht ein Thema besser als keins?}
\vers{Zu zweit war der Weg wirklich nicht weit.}
\vers{Alles vergeht, doch gemeinsam vergeht es schneller.}
\vers{Heureka! Das also sind Freunde!}
\end{strophe}
