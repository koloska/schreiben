\section{Der Lümmel sucht sie im Getümmel}
\stepcounter{ctrpoeminyear}
%\addcontentsline{toc}{section}{\thechapter.\arabic{ctrpoeminyear}\; Der Lümmel sucht sie im Getümmel}

\begin{strophe}
\vers{Zum Getümmel}
\vers{spricht der Lümmel,}
\vers{\ttquotl Willst du meine Braut mir sein?}
\vers{Komm wir wollen innen rein!}
\vers{Schön den Ausguss öffne mir,}
\vers{du wundervolles Trampeltier!}
\vers{Komm ich will dein Sklave sein.\ttquotr}
\vers{Dann der Frosche platzt herein,}
\vers{schlägt mit seiner Zunge auf}
\vers{den Lümmel mit Radau.}
\vers{Küsst die Braut in ihrer Not,}
\vers{jetzt springt sie immerzu hinfort.}
\end{strophe}
