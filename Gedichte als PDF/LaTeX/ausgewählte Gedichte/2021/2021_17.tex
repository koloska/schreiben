\section{Herzinfarkt auf dem Gehsteig}
\stepcounter{ctrpoeminyear}
%\addcontentsline{toc}{section}{\thechapter.\arabic{ctrpoeminyear}\; Herzinfarkt auf dem Gehsteig}

\begin{strophe}
\vers{Totenstill schreiten obskure Gestalten}
\vers{behände durch der Altstadt enge Gassen.}
\vers{Passanten stieben bei Notiz der Gestalt}
\vers{in alle Richtungen, als Düster und Kalt}
\vers{aus ihrem Gewand zu rinnen begann.}
\end{strophe}

\begin{strophe}
\vers{Vor einer bestimmten türmenden Person}
\vers{der Schatten sich auftürmte empor;}
\vers{weit über das lichte Firmament,}
\vers{sodass der ohnmächtige Mensch}
\vers{eingeschlossen wurde in eine Sphäre aus nichts.}
\end{strophe}

\begin{strophe}
\vers{Entblößt, nur von einem Faden umwickelt}
\vers{vernahm er loderndes Flüstern aus ferner Welt.}
\vers{Noch eh er erkannt um was er geraubt}
\vers{ward der Faden von ihm geschnitten,}
\vers{das Dunkel in die helle Tunika rekurriert.}
\end{strophe}

\begin{strophe}
\vers{Und ein lebloses Geschöpf lag dar,}
\vers{mitten auf dem verlassenen Trottoir.}
\end{strophe}
