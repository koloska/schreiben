\section{Kinderjäger}
\stepcounter{ctrpoeminyear}
%\addcontentsline{toc}{section}{\thechapter.\arabic{ctrpoeminyear} \; Kinderjäger}

\phantom{Dort}I.

\begin{strophe}
\vers{Draußen stehen sie,}
\vers{gekleidet wie Teufel.}
\vers{Man sieht sie zwar nie,}
\vers{doch sie laufen mit Beuteln.}
\end{strophe}

\begin{strophe}
\vers{Wohin sie wohl gehen?}
\vers{Man fragte sie nie.}
\vers{Warum sie dort stehen?}
\vers{Es ist wohl ein Spiel.}
\end{strophe}

\begin{strophe}
\vers{Wozu ihre Beutel?}
\vers{Na, das ist doch klar!}
\vers{Zum Stehlen der Kinder.}
\vers{Der Kinderlein Schar.}
\end{strophe}

II.

\begin{strophe}
\vers{Dort steh'n sie im Schatten}
\vers{an einander gereiht.}
\vers{Warten in der Kälte,}
\vers{ihre Blicke geh'n weit.}
\end{strophe}

\begin{strophe}
\vers{Wie lang sie wohl warten?}
\vers{So lang sich's ergibt.}
\vers{Werden sie fündig?}
\vers{Oder werden erwischt.}
\end{strophe}

\begin{strophe}
\vers{Keiner kennt sie,}
\vers{keiner weiß wer sie sind.}
\vers{Sie sind unter sich}
\vers{ganz allein in der Welt.}
\end{strophe}
