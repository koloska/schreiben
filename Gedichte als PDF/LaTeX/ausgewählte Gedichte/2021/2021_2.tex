\section{Bebende Welt}
\stepcounter{ctrpoeminyear}
%\addcontentsline{toc}{section}{\thechapter.\arabic{ctrpoeminyear} \; Bebende Welt}

\begin{strophe}
\vers{Eines stillen Tages die Stille verschwand}
\vers{als es aus den Tiefen zu beben begann.}
\vers{Es bebte so kräftig, es bebte so lang,}
\vers{ein jedes Kind auf Erden zu weinen begann.}
\end{strophe}

\begin{strophe}
\vers{Der Himmel stürzte, der Himmel zerbrach,}
\vers{wunde Lüfte in toten Lungen lag.}
\vers{Als Blitz und Donner schallend im Boden verschwand,}
\vers{tausende Seelen und Körper im Feuer verbrannt.}
\end{strophe}

\begin{strophe}
\vers{Als die Meere steigend das Land übernahm,}
\vers{Menschheit im Ängstewahn zu Grunde kam.}
\vers{Kein einziger blieb die Geschichte zu erzählen,}
\vers{ein jeder musste die Welt zu Ende gehen sehen.}
\end{strophe}
