\fancyhead[LO,RE]{Entwürfe}

\begin{strophe}
\vers{Finstre Mienen durch die Straßen grasen,}
\vers{Kein Lachen unter feuchten Nasen.}
\vers{Trocken tränen Tränen aus Augen aus,}
\vers{kein Wässerchen die Backe streift}
\vers{und doch hört man ein Heulen in der Welt,}
\vers{ein Gräulen, wie noch nie erwähnt.}
\end{strophe}
