\begin{strophe}
\vers{Die milchig gelbe Substanz im Glase}
\vers{mit feinsten Fäden ist geschmückt,}
\vers{die durch des Wassers Ruhe gleiten}
\vers{und wirbelnd heiter spiel'n vergnügt.}
\end{strophe}

\begin{strophe}
\vers{Der allzeit im Glas präsente Geruch}
\vers{sich froh in Synapsen zu verlieren versucht.}
\vers{Doch des Geschmacks zierliche Seite}
\vers{bleibt auf der Zunge noch eine Weile.}
\end{strophe}
