\begin{strophe}
\vers{Wie immer sitze ich in meinem Zimmer.}
\vers{Und bis ich nimmer wimmer,}
\vers{geht's mir immer, immer schlimmer.}
\end{strophe}

\begin{strophe}
\vers{Mit Etwas Mut,}
\vers{seh' ich mein Blut.}
\vers{Der Schmerz, er tut,}
\vers{dass es mir gut.}
\end{strophe}

\begin{strophe}
\vers{Mein Buch liegt auf dem Tresen.}
\vers{Das Tuch liegt auf dem Tresen.}
\vers{Heut will ich noch lesen!}
\vers{Putzen mit Tuch und Besen!}
\vers{-- Spät Nachts dann, beim verwesen,}
\vers{merke ich: Das Buch liegt ungelesen,}
\vers{Unbewegt liegt auch das Tuch, der Besen.}
\end{strophe}

\begin{strophe}
\vers{Ich bin nicht Faul.}
\vers{Ich bin nur Flau.}
\end{strophe}

\begin{strophe}
\vers{Das wisst ihr nicht.}
\vers{Das weiß nur ich.}

\vers{Mir geht's nicht gut.}
\vers{Mir geht's nicht gut.}
\end{strophe}

\begin{strophe}
\vers{Mir geht's nicht gut.}
\vers{Ich wünscht' mich tot.}
\end{strophe}
