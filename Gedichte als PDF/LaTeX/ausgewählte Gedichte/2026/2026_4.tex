\section{Große Wände}
\stepcounter{ctrpoeminyear}
%\addcontentsline{toc}{section}{\thechapter.\arabic{ctrpoeminyear} \; Große Wände}

\begin{strophe}
\vers{Große Wände um mich rum}
\vers{verschlucken jede Freude.}
\vers{Ich frage mich warum}
\vers{ausgerechnet heute.}
\end{strophe}

\begin{strophe}
\vers{Es pocht mein Herz}
\vers{vor lauter Schmerz}
\vers{und lautlos Schrei}
\vers{wird wieder frei.}
\end{strophe}

\begin{strophe}
\vers{Hoffnung verblasst}
\vers{und strömt aus meiner Seele.}
\vers{Denn mit aller Kraft fasst}
\vers{Verzweiflung meine Kehle.}
\end{strophe}

\begin{strophe}
\vers{Jeden Tag}
\vers{stirbt in mir drin}
\vers{ein Teil der bat}
\vers{ich möge finden einen Sinn}
\vers{um weiter noch zu leben.}
\end{strophe}
