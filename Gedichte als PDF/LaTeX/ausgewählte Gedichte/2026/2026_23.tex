\begin{strophe}
\vers{Regen tanzt auf uns herab,}
\vers{Gräßer fressen sich dran satt.}
\vers{Bäume schwanken hin und her,}
\vers{als ob sie gleich betrunten wär'n.}
\vers{Würmer kriechen aus ihr'm Haus,}
\vers{schauen in den Himmel auf,}
\vers{schauen auf die Erd' hinunter,}
\vers{alles Leben -- plötzlich munter.}
\vers{Pflanzen, Tiere und wir Menschen}
\vers{kennen plötzlich keine Grenzen.}
\vers{Schreiend rennt alles umher,}
\vers{als gebe es kein morgen mehr.}
\vers{Alles musiziert im Takt}
\vers{des regen Geplätschere Akt.}
\vers{So dreht die Welt sich immer weiter,}
\vers{wenn wieder Wolken weinen heiter.}
\end{strophe}
