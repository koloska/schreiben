\begin{strophe}
\vers{Schläfst du, Kindlein? Schlaf in Ruh'.}
\vers{Halte deine Äuglein zu.}
\vers{Der Tag wird Schwierigkeiten bringen.}
\vers{Möchtest du ihn nicht besingen?}
\vers{Dein schnarchend Lied wird nicht lang halten.}
\vers{Bald schon wird der Morgen spalten}
\vers{die Nacht vom Tag, sodass du wieder}
\vers{öffnen musst der Augen Lider,}
\vers{strecken tust die langen Glieder,}
\vers{obgleich es dir wohl wäre lieber}
\vers{du hättest noch ne Stund' geschlafen.}
\vers{Doch jetzt musst du dich endlich raffen!}
\vers{Der Tag wartet ja nicht auf dich,}
\vers{du musst ihn jagen, was ein Pech.}
\vers{Stehe auf, faule Schabracke,}
\vers{ziehe an die blöde Jacke,}
\vers{tret' nach draußen}
\vers{in aller Ruh',}
\vers{denn nebenan schläft doch noch Nachbar Gu.}
\end{strophe}
