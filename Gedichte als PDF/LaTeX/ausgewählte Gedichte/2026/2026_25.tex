\begin{strophe}
\vers{Kleidung, Leiter, Wechselbirne}
\vers{Apparat, der dreht sich}
\vers{Der dreht die Birne}
\vers{Die Birne dreht sich}
\vers{Sie dreht und sie dreht und sie dreht immer weiter}
\vers{Sie dreht bis sie anhält}
\vers{Dann gehts nicht mehr weiter.}
\vers{Dann gehts weiter}
\vers{Die nächste Birne}
\vers{Eine raus, eine rein}
\vers{Und drehen}
\vers{Und drehen}
\vers{Und drehen bis Stopp!}
\vers{Stopp!}
\vers{Jetzt dreht sie nicht mehr}
\vers{Sie hat gedreht und jetzt ist sie drin}
\vers{Bis alle drin sind}
\vers{Drehen}
\vers{Und drin}
\vers{Drehen}
\vers{Und drin}
\vers{Und drin und drin und drin und drin}
\vers{Alle sind drin}
\vers{Ende der Geschicht}
\vers{Aus mit der Arbeit}
\vers{Die alten zertrümmern}
\vers{Ei, ja das macht Spaß}
\vers{Hau der Birne die Birne weg!}
\vers{Jetzt Splitter}
\vers{Splitterfasernackt}
\vers{Splitterfasernackt sind die Glühdrähte}
\vers{Sie glühen nicht mehr}
\vers{Sie sind ausgebrannt. und wurden ausgetauscht}
\vers{Was ein Schmaus.}
\end{strophe}
