\begin{strophe}
\vers{Gedichte schreiben ist so toll}
\vers{weil es mich rand voll}
\vers{mit Emotionen füllt,}
\vers{die meine Leere übertönen.}
\end{strophe}

\begin{strophe}
\vers{Mein Herzlein springt}
\vers{dann auf und ab}
\vers{und stellt}
\vers{die richt'gen Fragen.}
\end{strophe}

\begin{strophe}
\vers{Gedichte schreiben ist so top}
\vers{weil es meinen Kopf}
\vers{mit Texten füllt,}
\vers{die all meine Gedanken überragen.}
\end{strophe}

\begin{strophe}
\vers{Komm, noch eins schreib}
\vers{in der Freizeit,}
\vers{von der ich grad}
\vers{so viel von habe.}
\end{strophe}
