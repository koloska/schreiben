\begin{strophe}
\vers{Tränen flossen wie aus Beton.}
\vers{Nie kam der Henker, obgleich er beschwor'n.}
\vers{Kreischender Gesang aus dem Sallon.}
\vers{Dort kam der Retter, als ob er verschwor'n.}
\end{strophe}

\begin{strophe}
\vers{Der Retter, ein Engel aus glänzendem Wind,}
\vers{umarmte völligst das pechschwarze Kind.}
\vers{Seiner Flügel Dornen fuhr'n in sein Kleid.}
\vers{Trüb Kindlein ward stechender Schmerzen sehr Leid.}
\vers{Doch bald, nach Stunden keuchenden Spuks,}
\vers{als durch Gezwitscher Nacht mit dem Tag verwuchs,}
\vers{verschwand dank Saugen der Dornen die schwarze Tinktur,}
\vers{der Fluch wich allmählich, sonach Engels Schwur.}
\vers{Das Kind, jetzt Träger der Helligkeit, lies funkelnd}
\vers{seinen Geist erstrahlen, der sonst so verdunkelt.}
\end{strophe}

\begin{strophe}
\vers{Es war ihm nun klar. Der Henker war fort.}
\vers{Um seines Lebens Kerze war es nicht mehr besorgt.}
\end{strophe}
