\section{Blümchen}
\stepcounter{ctrpoeminyear}
%\addcontentsline{toc}{section}{\thechapter.\arabic{ctrpoeminyear} \; Blümchen}

\begin{strophe}
\vers{Blümchen, du heilige Pracht der Natur,}
\vers{wo du tanzet tritt stets Leben und Liebe hervor.}
\vers{Blümchen, oh wie dein Glanz die Seele erhellt.}
\vers{Dein Kleid schmückt den Garten der sterblichen Welt.}
\end{strophe}

\begin{strophe}
\vers{Nebst Bäumen und Gräßern kämpfst du dich ans Licht}
\vers{und zauberst ein Lächeln in jedes Gesicht.}
\vers{Du schneidest mit Farbe in Monotonie}
\vers{und bastelst ein Spiel aus Komöd' und Tragie.}
\end{strophe}

\begin{strophe}
\vers{Das Lied, das du singest, stimmt freudig die Wiesen}
\twolinevers{und Felder, Täler, Wälder und Orte von denen wir nicht}{wissen.}
\vers{Bevor du verwelkst spendest du unzählig viel Freude,}
\vers{Bleib bei mir, geh nicht, verzeih meine Reue.}
\end{strophe}
